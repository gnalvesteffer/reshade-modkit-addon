cmake_minimum_required(VERSION 3.23)
project(ReShadeModKitAddon VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Use a Windows compiler or mingw64.cmake")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(HOOK_SOURCES vendor/minhook/src/hook.c vendor/minhook/src/buffer.c vendor/minhook/src/trampoline.c)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  list(APPEND HOOK_SOURCES vendor/minhook/src/hde/hde64.c)
  set(ADDON_SUFFIX .addon64)
else()
  list(APPEND HOOK_SOURCES vendor/minhook/src/hde/hde32.c)
  set(ADDON_SUFFIX .addon32)
endif()
add_library(ModKit MODULE source/modkit.cpp source/stb.c ${HOOK_SOURCES})
target_include_directories(ModKit PRIVATE source vendor/reshade vendor/imgui vendor/stb vendor/minhook/include vendor/dxc)
target_compile_definitions(ModKit PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN STBI_NO_STDIO STBI_WRITE_NO_STDIO)
target_link_libraries(ModKit PRIVATE d3dcompiler dxguid)
set_target_properties(ModKit PROPERTIES PREFIX "" SUFFIX "${ADDON_SUFFIX}")
if(MINGW)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/compat")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compat/Windows.h" "#include <windows.h>\n")
  target_include_directories(ModKit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/compat")
  target_link_options(ModKit PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
install(TARGETS ModKit LIBRARY DESTINATION . RUNTIME DESTINATION .)
